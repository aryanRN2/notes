\section{Module 1: Foundations of Metric Spaces and Semi-Metrics}

A metric space is a mathematical abstraction that formalizes and generalizes the intuitive geometric concept of distance between points in any set.

\subsection{Definition of a Metric Space}

\begin{defbox}{Definition: Metric and Metric Space}
Let $X$ be a non-empty set. A mapping $d: X \times X \to \mathbb{R}$ is called a \textbf{metric} (or distance function) on $X$ if and only if it satisfies the following four fundamental axioms for all elements $x, y, z \in X$:
\begin{enumerate}
    \item[\textbf{[M1]}] \textbf{Non-negativity:}
    \begin{equation}
        d(x, y) \ge 0, \quad \forall x, y \in X
    \end{equation}
    
    \item[\textbf{[M2]}] \textbf{Identity of Indiscernibles (Definiteness):}
    \begin{equation}
        d(x, y) = 0 \iff x = y
    \end{equation}
    
    \item[\textbf{[M3]}] \textbf{Symmetry:}
    \begin{equation}
        d(x, y) = d(y, x), \quad \forall x, y \in X
    \end{equation}
    
    \item[\textbf{[M4]}] \textbf{Triangle Inequality:}
    \begin{equation}
        d(x, y) \le d(x, z) + d(z, y), \quad \forall x, y, z \in X
    \end{equation}
\end{enumerate}
If $d$ is a metric on $X$, then the ordered pair $(X, d)$ is called a \textbf{Metric Space}, and the non-negative real number $d(x, y)$ is called the \textbf{distance between $x$ and $y$}.
\end{defbox}

\subsection{Semi-Metric (Pseudo-Metric)}

\begin{defbox}{Definition: Semi-Metric / Pseudo-Metric}
A mapping $d: X \times X \to \mathbb{R}$ is called a \textbf{semi-metric} (or \textbf{pseudo-metric}) if and only if it satisfies:
\begin{enumerate}
    \item[\textbf{[M1]}] $d(x, y) \ge 0, \quad \forall x, y \in X$.
    \item[\textbf{[M2']}] $d(x, x) = 0, \quad \forall x \in X$ (Note: $d(x, y) = 0$ does \textbf{not} require $x = y$).
    \item[\textbf{[M3]}] $d(x, y) = d(y, x), \quad \forall x, y \in X$.
    \item[\textbf{[M4]}] $d(x, y) \le d(x, z) + d(z, y), \quad \forall x, y, z \in X$.
\end{enumerate}
\textbf{Remark:} Every metric is a semi-metric, but a semi-metric is generally \textbf{not} a metric because distinct points may have zero distance.
\end{defbox}

\begin{exbox}{Example: A Semi-Metric that is NOT a Metric}
\textbf{Question:} Consider the mapping $d: \mathbb{R} \times \mathbb{R} \to \mathbb{R}$ defined by:
\begin{equation}
    d(x, y) = |x^2 - y^2|, \quad \forall x, y \in \mathbb{R}
\end{equation}
Show that $d$ is a semi-metric on $\mathbb{R}$, but is \textbf{not} a metric.

\textbf{Verification of Axioms:}
\begin{enumerate}
    \item \textbf{[M1] Non-negativity:} Since absolute value is always non-negative, $d(x, y) = |x^2 - y^2| \ge 0, \; \forall x, y \in \mathbb{R}$.
    
    \item \textbf{[M3] Symmetry:} $d(x, y) = |x^2 - y^2| = |-(y^2 - x^2)| = |y^2 - x^2| = d(y, x)$.
    
    \item \textbf{[M4] Triangle Inequality:} For any $x, y, z \in \mathbb{R}$:
    \begin{align}
        d(x, y) &= |x^2 - y^2| = |(x^2 - z^2) + (z^2 - y^2)| \notag \\
        &\le |x^2 - z^2| + |z^2 - y^2| = d(x, z) + d(z, y)
    \end{align}
    
    \item \textbf{[M2] Failure of Indiscernibles:}
    \begin{align}
        d(x, y) = 0 &\iff |x^2 - y^2| = 0 \iff x^2 = y^2 \notag \\
        &\iff x = y \quad \text{or} \quad x = -y
    \end{align}
    For instance, choose $x = 2$ and $y = -2$:
    $$d(2, -2) = |2^2 - (-2)^2| = |4 - 4| = 0$$
    However, $2 \neq -2$. Therefore, condition $[M2]$ fails.
\end{enumerate}
\textbf{Conclusion:} $d(x, y) = |x^2 - y^2|$ is a \textbf{semi-metric} (pseudo-metric), but \textbf{not a metric} on $\mathbb{R}$. \qed
\end{exbox}

\subsection{Standard Metrics on Euclidean Spaces}

\begin{formulabox}{Standard Metrics on $\mathbb{R}^2$}
Let $x = (x_1, y_1)$ and $y = (x_2, y_2)$ be points in $\mathbb{R}^2$.
\begin{enumerate}
    \item \textbf{Euclidean Metric ($d_2$):}
    \begin{equation}
        d_2(x, y) = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}
    \end{equation}
    
    \item \textbf{Taxicab / Manhattan Metric ($d_1$):}
    \begin{equation}
        d_1(x, y) = |x_1 - x_2| + |y_1 - y_2|
    \end{equation}
    
    \item \textbf{Maximum / Chebyshev Metric ($d_\infty$):}
    \begin{equation}
        d_\infty(x, y) = \max\left( |x_1 - x_2|, |y_1 - y_2| \right)
    \end{equation}
\end{enumerate}
\end{formulabox}

\subsection{Discrete Metric Space}

\begin{defbox}{Definition: Discrete Metric Space}
Let $X$ be any arbitrary non-empty set. The \textbf{discrete metric} on $X$ is defined as:
\begin{equation}
    d(x, y) = 0 \quad \text{if } x = y, \qquad \text{and} \qquad d(x, y) = 1 \quad \text{if } x \neq y
\end{equation}
The pair $(X, d)$ is called a \textbf{Discrete Metric Space}. In this space, every single point is isolated at a distance of 1 from every other point.
\end{defbox}

\subsection{Matrix Spaces as Metric Spaces}

Let $M_{m \times n}(\mathbb{R})$ denote the vector space of all $m \times n$ real matrices.

\begin{defbox}{Definition: Frobenius Metric on Matrix Space}
For any matrices $A, B \in M_{m \times n}(\mathbb{R})$, the \textbf{Frobenius distance} is defined by:
\begin{equation}
    d_F(A, B) = \|A - B\|_F = \sqrt{\sum_{i=1}^m \sum_{j=1}^n (a_{ij} - b_{ij})^2} = \sqrt{\operatorname{Tr}\left((A-B)^T(A-B)\right)}
\end{equation}
\textbf{Properties:}
\begin{itemize}
    \item $d_F(A, B) \ge 0$ and $d_F(A, B) = 0 \iff A = B$.
    \item $d_F(A, B) = d_F(B, A)$.
    \item $d_F(A, B) \le d_F(A, C) + d_F(C, B)$ (follows from Cauchy-Schwarz on matrix trace).
    \item $(M_{m \times n}(\mathbb{R}), d_F)$ is isometrically isomorphic to the Euclidean space $(\mathbb{R}^{mn}, d_2)$ and is therefore a \textbf{complete metric space}.
\end{itemize}
\end{defbox}

\section{Module 2: Open Spheres, Neighborhoods, and Interior Points}

\subsection{Open Spheres (Open Balls)}

\begin{defbox}{Definition: Open Sphere / Open Ball}
Let $(X, d)$ be a metric space, $x_0 \in X$, and $r > 0$ be a positive real number. The \textbf{open sphere} (or open ball) centered at $x_0$ of radius $r$ is the subset:
\begin{equation}
    S_r(x_0) = B(x_0, r) = \{ x \in X \mid d(x, x_0) < r \}
\end{equation}
The \textbf{closed sphere} centered at $x_0$ of radius $r$ is:
\begin{equation}
    \bar{S}_r(x_0) = \bar{B}(x_0, r) = \{ x \in X \mid d(x, x_0) \le r \}
\end{equation}
\end{defbox}

\subsection{Geometric Shapes of Open Spheres in Different Metrics}

\begin{exbox}{Examples: Open Spheres with $x_0 = (0,0)$ and $r = 1$}
\begin{enumerate}
    \item \textbf{Usual Metric on $\mathbb{R}$ with $x_0 = 0, r = 1$:}
    $$S_1(0) = \{ x \in \mathbb{R} \mid d(x, 0) < 1 \} = \{ x \in \mathbb{R} \mid |x| < 1 \} = \mathbf{(-1, 1)} \quad \text{(Open Interval)}$$
    
    \item \textbf{Euclidean Metric $(\mathbb{R}^2, d_2)$ with $x_0 = (0,0), r = 1$:}
    $$S_1(0,0) = \{ (x, y) \in \mathbb{R}^2 \mid \sqrt{x^2 + y^2} < 1 \} = \{ (x, y) \mid \mathbf{x^2 + y^2 < 1} \} \quad \text{(Circular Disk)}$$
    
    \item \textbf{Taxicab Metric $(\mathbb{R}^2, d_1)$ with $x_0 = (0,0), r = 1$:}
    $$S_1(0,0) = \{ (x, y) \in \mathbb{R}^2 \mid \mathbf{|x| + |y| < 1} \} \quad \text{(Rotated Square / Diamond)}$$
    
    \item \textbf{Maximum Metric $(\mathbb{R}^2, d_\infty)$ with $x_0 = (0,0), r = 1$:}
    $$S_1(0,0) = \{ (x, y) \in \mathbb{R}^2 \mid \max(|x|, |y|) < 1 \} = \mathbf{(-1, 1) \times (-1, 1)} \quad \text{(Axis-aligned Square)}$$
    
    \item \textbf{Discrete Metric Space $(X, d)$:}
    $$S_r(x_0) = \{x_0\} \quad \text{if } r \le 1, \qquad \text{and} \qquad S_r(x_0) = X \quad \text{if } r > 1$$
\end{enumerate}
\end{exbox}

\subsection{Neighborhoods and Interior Points}

\begin{defbox}{Definitions: Neighborhood and Open Neighborhood}
Let $(X, d)$ be a metric space and $x \in X$.
\begin{itemize}
    \item A subset $N \subseteq X$ is called a \textbf{neighborhood of $x$} if and only if there exists an open sphere $S_r(x)$ centered at $x$ with radius $r > 0$ such that:
    \begin{equation}
        S_r(x) \subseteq N
    \end{equation}
    \item If the neighborhood $N$ is itself an open set, it is called an \textbf{open neighborhood} of $x$.
\end{itemize}
\end{defbox}

\begin{defbox}{Definitions: Interior Point and Interior of a Set}
Let $G$ be a subset of a metric space $(X, d)$.
\begin{itemize}
    \item A point $x \in G$ is called an \textbf{interior point} of $G$ if and only if there exists an open sphere $S_r(x)$ with $r > 0$ such that:
    \begin{equation}
        S_r(x) \subseteq G
    \end{equation}
    \item The set of all interior points of $G$ is called the \textbf{interior of $G$}, denoted by $G^\circ$ or $\operatorname{Int}(G)$.
    \item \textbf{Theorem:} A subset $G$ is an \textbf{open set} if and only if every point of $G$ is an interior point of $G$ ($G = G^\circ$).
\end{itemize}
\end{defbox}

\section{Module 3: Topology of Open Sets and Fundamental Theorems}

\subsection{Open Sets in a Metric Space}

\begin{defbox}{Definition: Open Set}
A subset $G$ of a metric space $(X, d)$ is called an \textbf{open set} if and only if for each point $x \in G$, there exists a real number $r > 0$ such that:
\begin{equation}
    S_r(x) \subseteq G
\end{equation}
\end{defbox}

\subsection{Fundamental Theorems on Open Sets with Complete Proofs}

\begin{exbox}{Theorem 1: The Empty Set $\emptyset$ and Full Space $X$ are Open}
\textbf{Theorem:} In any metric space $(X, d)$, the empty set $\emptyset$ and the full space $X$ are open sets.

\textbf{Proof:}
\begin{enumerate}
    \item \textbf{For the empty set $\emptyset$:} A set $G$ is open if every $x \in G$ has an open ball contained in $G$. Since $\emptyset$ contains no elements, the condition holds vacuously. Thus, $\emptyset$ is open.
    \item \textbf{For the full space $X$:} Let $x \in X$ be arbitrary. Choose any radius $r > 0$. By definition, the open sphere $S_r(x) = \{y \in X \mid d(y, x) < r\} \subseteq X$. Hence every point of $X$ is an interior point of $X$, so $X$ is open. \qed
\end{enumerate}
\end{exbox}

\begin{exbox}{Theorem 2: Every Open Sphere is an Open Set}
\textbf{Theorem:} In a metric space $(X, d)$, every open sphere $S_r(x_0)$ is an open set.

\textbf{Proof:}
\begin{enumerate}
    \item Let $S_r(x_0)$ be an open sphere centered at $x_0 \in X$ with radius $r > 0$.
    \item Let $x$ be an arbitrary point in $S_r(x_0)$. Then by definition:
    \begin{equation}
        d(x, x_0) < r \implies r - d(x, x_0) > 0
    \end{equation}
    \item Choose the radius $\rho = r - d(x, x_0) > 0$.
    \item \textbf{Claim:} $S_\rho(x) \subseteq S_r(x_0)$.
    \item To prove the claim, let $y \in S_\rho(x)$. Then $d(y, x) < \rho$.
    \item By the triangle inequality $[M4]$ on the metric $d$:
    \begin{align}
        d(y, x_0) &\le d(y, x) + d(x, x_0) \notag \\
        &< \rho + d(x, x_0) \notag \\
        &= (r - d(x, x_0)) + d(x, x_0) \notag \\
        &= r
    \end{align}
    \item Since $d(y, x_0) < r$, it follows that $y \in S_r(x_0)$.
    \item Therefore, $S_\rho(x) \subseteq S_r(x_0)$.
    \item Since $x \in S_r(x_0)$ was arbitrary, every point in $S_r(x_0)$ is an interior point.
    \item Hence, $S_r(x_0)$ is an \textbf{open set}. \qed
\end{enumerate}
\end{exbox}

\begin{exbox}{Theorem 3: Arbitrary Union of Open Sets is Open}
\textbf{Theorem:} In a metric space $(X, d)$, the union of an arbitrary collection of open sets is an open set.

\textbf{Proof:}
\begin{enumerate}
    \item Let $\{G_\lambda \mid \lambda \in \Lambda\}$ be an arbitrary collection of open sets in $(X, d)$, where $\Lambda$ is an arbitrary index set.
    \item Let $G = \bigcup_{\lambda \in \Lambda} G_\lambda$.
    \item If $G = \emptyset$, then $G$ is open by Theorem 1.
    \item If $G \neq \emptyset$, let $x \in G$.
    \item By definition of union, $x \in G_{\lambda_0}$ for some specific index $\lambda_0 \in \Lambda$.
    \item Since $G_{\lambda_0}$ is an open set, there exists a radius $r > 0$ such that:
    \begin{equation}
        S_r(x) \subseteq G_{\lambda_0}
    \end{equation}
    \item Since $G_{\lambda_0} \subseteq \bigcup_{\lambda \in \Lambda} G_\lambda = G$, it follows that:
    \begin{equation}
        S_r(x) \subseteq G
    \end{equation}
    \item Since $x \in G$ was chosen arbitrarily, every point of $G$ is an interior point.
    \item Hence, $G = \bigcup_{\lambda \in \Lambda} G_\lambda$ is an \textbf{open set}. \qed
\end{enumerate}
\end{exbox}

\begin{exbox}{Theorem 4: Finite Intersection of Open Sets is Open}
\textbf{Theorem:} In a metric space $(X, d)$, the intersection of a finite number of open sets is an open set.

\textbf{Proof:}
\begin{enumerate}
    \item Let $G_1, G_2, \dots, G_n$ be a finite collection of open sets in $(X, d)$.
    \item Let $G = \bigcap_{i=1}^n G_i$.
    \item If $G = \emptyset$, then $G$ is open by Theorem 1.
    \item If $G \neq \emptyset$, let $x \in G$.
    \item By definition of intersection, $x \in G_i$ for every $i \in \{1, 2, \dots, n\}$.
    \item Since each $G_i$ is open, there exist positive real radii $r_1, r_2, \dots, r_n > 0$ such that:
    \begin{equation}
        S_{r_i}(x) \subseteq G_i, \quad \forall i \in \{1, 2, \dots, n\}
    \end{equation}
    \item Define $r = \min \{r_1, r_2, \dots, r_n\}$. Since it is the minimum of a \textbf{finite} collection of strictly positive numbers, $r > 0$.
    \item Since $r \le r_i$ for each $i$, we have $S_r(x) \subseteq S_{r_i}(x) \subseteq G_i$ for all $i = 1, 2, \dots, n$.
    \item Therefore:
    \begin{equation}
        S_r(x) \subseteq \bigcap_{i=1}^n G_i = G
    \end{equation}
    \item Since $x \in G$ was arbitrary, $G = \bigcap_{i=1}^n G_i$ is an \textbf{open set}. \qed
\end{enumerate}
\end{exbox}

\begin{formulabox}{Important Counterexample: Infinite Intersection of Open Sets May NOT be Open}
In $(\mathbb{R}, |\cdot|)$, consider the infinite family of open intervals:
\begin{equation}
    G_n = \left( -\frac{1}{n}, \frac{1}{n} \right), \quad n \in \mathbb{N}
\end{equation}
Each $G_n$ is an open set in $\mathbb{R}$. However, their infinite intersection is:
\begin{equation}
    \bigcap_{n=1}^\infty G_n = \bigcap_{n=1}^\infty \left( -\frac{1}{n}, \frac{1}{n} \right) = \{0\}
\end{equation}
The singleton set $\{0\}$ is a \textbf{closed set} and is \textbf{not open} in $(\mathbb{R}, |\cdot|)$, because for any $r > 0$, the open ball $S_r(0) = (-r, r) \not\subseteq \{0\}$.
\end{formulabox}

\section{Module 4: Limit Points, Derived Sets, and Closed Sets}

\subsection{Limit Points (Accumulation Points)}

\begin{defbox}{Definition: Limit Point of a Set}
Let $(X, d)$ be a metric space and $A \subseteq X$. A point $p \in X$ is called a \textbf{limit point} (or \textbf{accumulation point} / cluster point) of $A$ if and only if every neighborhood $N_p$ of $p$ contains at least one point of $A$ different from $p$:
\begin{equation}
    \left( N_p \setminus \{p\} \right) \cap A \neq \emptyset
\end{equation}
Equivalently, for every $r > 0$:
\begin{equation}
    \left( S_r(p) \setminus \{p\} \right) \cap A \neq \emptyset
\end{equation}
\end{defbox}

\subsection{Derived Sets and Examples}

\begin{defbox}{Definition: Derived Set $A'$}
The set of all limit points of a subset $A$ is called the \textbf{derived set} of $A$, denoted by $A'$ or $d(A)$.
\end{defbox}

\begin{exbox}{Solved Examples on Derived Sets}
\begin{enumerate}
    \item \textbf{Harmonic Sequence Set in $(\mathbb{R}, |\cdot|)$:}
    Let $A = \left\{ 1, \frac{1}{2}, \frac{1}{3}, \dots, \frac{1}{n}, \dots \right\}$.
    The only accumulation point is $0$, because for every $\varepsilon > 0$, $(-\varepsilon, \varepsilon) \cap A$ contains infinitely many terms $\frac{1}{n}$. None of the elements $\frac{1}{n} \in A$ are limit points.
    $$\mathbf{A' = \{0\}}$$
    
    \item \textbf{Open Interval in $(\mathbb{R}, |\cdot|)$:}
    Let $A = (0, 1)$. Every point in $(0, 1)$ as well as the endpoints $0$ and $1$ are limit points:
    $$\mathbf{A' = [0, 1]}$$
    
    \item \textbf{Topologist's Sine Curve in $\mathbb{R}^2$:}
    Consider the subset $A = \left\{ (x, y) \in \mathbb{R}^2 \mid y = \sin\left(\frac{1}{x}\right), \; x > 0 \right\}$.
    As $x \to 0^+$, $\sin(1/x)$ oscillates between $-1$ and $+1$ infinitely often.
    The derived set $A'$ contains the curve itself plus the entire vertical segment on the $y$-axis:
    $$\mathbf{A' = A \cup \left( \{0\} \times [-1, 1] \right)}$$
\end{enumerate}
\end{exbox}

\subsection{Closed Sets in a Metric Space}

\begin{defbox}{Definition: Closed Set}
Let $(X, d)$ be a metric space. A subset $F \subseteq X$ is called a \textbf{closed set} if and only if its complement $F^c = X \setminus F$ is an open set in $(X, d)$.
\end{defbox}

\begin{exbox}{Examples of Closed Sets}
\begin{enumerate}
    \item \textbf{Singleton Sets in $(\mathbb{R}, |\cdot|)$:}
    For any $x \in \mathbb{R}$, the singleton $\{x\}$ is closed because its complement:
    $$\{x\}^c = (-\infty, x) \cup (x, \infty)$$
    is the union of two open intervals (open sets), hence open. Thus, \textbf{every singleton set in a metric space is closed}.
    
    \item \textbf{Closed Intervals in $(\mathbb{R}, |\cdot|)$:}
    $[a, b]$ is closed because $[a, b]^c = (-\infty, a) \cup (b, \infty)$ is open.
\end{enumerate}
\end{exbox}

\subsection{Topological Properties of Closed Sets}

\begin{formulabox}{Duality Properties of Closed Sets (De Morgan's Laws)}
\begin{enumerate}
    \item The empty set $\emptyset$ and the whole space $X$ are \textbf{closed sets} (since $\emptyset^c = X$ and $X^c = \emptyset$ are open).
    \item The \textbf{intersection of an arbitrary collection} of closed sets is closed:
    $$\left( \bigcap_{\lambda \in \Lambda} F_\lambda \right)^c = \bigcup_{\lambda \in \Lambda} F_\lambda^c \quad \text{(Union of open sets is open)}$$
    \item The \textbf{union of a finite number} of closed sets is closed:
    $$\left( \bigcup_{i=1}^n F_i \right)^c = \bigcap_{i=1}^n F_i^c \quad \text{(Finite intersection of open sets is open)}$$
\end{enumerate}
\end{formulabox}

\subsection{Theorem: Limit Point Characterization of Closed Sets}

\begin{exbox}{Theorem: Closed Sets Contain All Their Limit Points}
\textbf{Theorem:} In a metric space $(X, d)$, a subset $F \subseteq X$ is closed if and only if $F$ contains all its limit points ($F' \subseteq F$).

\textbf{Proof:}
\begin{enumerate}
    \item \textbf{($\implies$) Necessary Condition:} Assume $F$ is closed. We must show $F' \subseteq F$.
    \begin{itemize}
        \item Let $p \in F^c = X \setminus F$.
        \item Since $F$ is closed, $F^c$ is an open set.
        \item By definition of an open set, there exists an open ball $S_r(p)$ such that $S_r(p) \subseteq F^c = X \setminus F$.
        \item This implies $S_r(p) \cap F = \emptyset \implies (S_r(p) \setminus \{p\}) \cap F = \emptyset$.
        \item Therefore, $p$ is \textbf{not} a limit point of $F$ ($p \notin F'$).
        \item By contrapositive, if $p \in F'$, then $p \in F$. Thus $F' \subseteq F$.
    \end{itemize}
    
    \item \textbf{($\impliedby$) Sufficient Condition:} Assume $F' \subseteq F$. We must show $F$ is closed (i.e., $F^c$ is open).
    \begin{itemize}
        \item Let $x \in F^c = X \setminus F$.
        \item Since $F' \subseteq F$, we have $x \notin F'$.
        \item Since $x$ is not a limit point of $F$, there exists an open sphere $S_r(x)$ such that:
        $$(S_r(x) \setminus \{x\}) \cap F = \emptyset$$
        \item Since $x \notin F$, we have $S_r(x) \cap F = \emptyset$, which implies $S_r(x) \subseteq X \setminus F = F^c$.
        \item Since $x \in F^c$ was arbitrary, every point of $F^c$ is an interior point of $F^c$.
        \item Thus $F^c$ is open, which proves that $F$ is \textbf{closed}. \qed
    \end{itemize}
\end{enumerate}
\end{exbox}

\subsection{Theorem: Every Derived Set in a Metric Space is Closed}

\begin{exbox}{Theorem: The Derived Set $F'$ is Closed in $(X, d)$}
\textbf{Theorem:} Let $(X, d)$ be a metric space and let $F \subseteq X$. Then the derived set $F'$ is a closed set in $(X, d)$ (that is, $(F')' \subseteq F'$).

\textbf{Proof:}
\begin{enumerate}
    \item We need to show that $(F')' \subseteq F'$, which by the limit point characterization means $F'$ is closed.
    \item If $(F')' = \emptyset$, then $(F')' \subseteq F'$ holds trivially.
    \item Now assume $(F')' \neq \emptyset$ and let $x \in (F')'$. That is, $x$ is a limit point of $F'$.
    \item We must prove that $x \in F'$ (i.e., $x$ is a limit point of $F$).
    \item Let $S_r(x)$ be any open sphere centered at $x$ with arbitrary radius $r > 0$.
    \item Since $x \in (F')'$, every open sphere centered at $x$ contains at least one point of $F'$ distinct from $x$. Thus, there exists a point:
    \begin{equation}
        y_r \in (S_r(x) \setminus \{x\}) \cap F'
    \end{equation}
    \item Since $y_r \in S_r(x)$, we have $d(x, y_r) < r \implies \rho = r - d(x, y_r) > 0$.
    \item Consider the open sphere $S_\rho(y_r)$ centered at $y_r$ with radius $\rho$.
    \item By the Fundamental Theorem on Open Balls, $S_\rho(y_r) \subseteq S_r(x)$.
    \item Furthermore, since $y_r \in F'$, $y_r$ is a limit point of $F$. Therefore, the open sphere $S_\rho(y_r)$ must contain a point $z_r \in F$ distinct from $y_r$:
    \begin{equation}
        z_r \in (S_\rho(y_r) \setminus \{y_r\}) \cap F
    \end{equation}
    \item Since $z_r \in S_\rho(y_r) \subseteq S_r(x)$, we have $z_r \in S_r(x)$ and $z_r \in F$.
    \item By choosing $\rho_1 = \min(\rho, d(x, y_r)) > 0$, we can strictly guarantee that $z_r \neq x$.
    \item Therefore, for every $r > 0$, the punctured open sphere $(S_r(x) \setminus \{x\})$ contains a point $z_r \in F$:
    \begin{equation}
        (S_r(x) \setminus \{x\}) \cap F \neq \emptyset
    \end{equation}
    \item This precisely proves that $x$ is a limit point of $F$, so $x \in F'$.
    \item Since $x \in (F')' \implies x \in F'$, we have $(F')' \subseteq F'$.
    \item Hence, the derived set $F'$ is a \textbf{closed set}. \qed
\end{enumerate}
\end{exbox}

\subsection{Closure of a Set and Its Topological Properties}

\begin{defbox}{Definition: Closure of a Set ($\overline{A}$)}
Let $(X, d)$ be a metric space and $A \subseteq X$. The \textbf{closure} of $A$, denoted by $\overline{A}$ or $\operatorname{Cl}(A)$, is defined as the union of the set $A$ and its derived set $A'$:
\begin{equation}
    \overline{A} = A \cup A'
\end{equation}
A point $x \in \overline{A}$ is called an \textbf{adherent point} (or closure point) of $A$. That is, every open sphere $S_r(x)$ intersects $A$ ($S_r(x) \cap A \neq \emptyset$).
\end{defbox}

\begin{formulabox}{Theorem: Fundamental Properties of Closure}
Let $(X, d)$ be a metric space and $A, B \subseteq X$. The closure operator satisfies:
\begin{enumerate}
    \item[\textbf{(i)}] $\overline{A}$ is a \textbf{closed set} in $(X, d)$.
    \item[\textbf{(ii)}] \textbf{Monotonicity:} If $A \subseteq B$, then $\overline{A} \subseteq \overline{B}$.
    \item[\textbf{(iii)}] \textbf{Characterization:} $A$ is a closed set if and only if $\overline{A} = A$.
    \item[\textbf{(iv)}] \textbf{Minimality:} $\overline{A}$ is the \textbf{smallest closed set} containing $A$, i.e.,
    $$\overline{A} = \bigcap \{ F \subseteq X \mid A \subseteq F \text{ and } F \text{ is closed} \}$$
    \item[\textbf{(v)}] \textbf{Finite Additivity:} $\overline{A \cup B} = \overline{A} \cup \overline{B}$.
    \item[\textbf{(vi)}] \textbf{Intersection Sub-additivity:} $\overline{A \cap B} \subseteq \overline{A} \cap \overline{B}$.
    \item[\textbf{(vii)}] \textbf{Idempotence (Kuratowski Axiom):} $\overline{\overline{A}} = \overline{A}$.
\end{enumerate}
\end{formulabox}

\subsubsection{Complete Rigorous Proofs for Closure Properties}

\begin{exbox}{Proof of Property (i): $\overline{A}$ is a Closed Set}
\textbf{To Prove:} $(\overline{A})^c$ is an open set.
\begin{enumerate}
    \item By definition, $\overline{A} = A \cup A'$. Using De Morgan's Law:
    \begin{equation}
        (\overline{A})^c = (A \cup A')^c = A^c \cap (A')^c
    \end{equation}
    \item Let $x \in (\overline{A})^c = A^c \cap (A')^c$. Then $x \in A^c$ and $x \in (A')^c$.
    \item Since $x \in (A')^c \implies x \notin A'$, $x$ is \textbf{not} a limit point of $A$.
    \item Therefore, there exists $r_1 > 0$ such that:
    \begin{equation}
        (S_{r_1}(x) \setminus \{x\}) \cap A = \emptyset
    \end{equation}
    Since $x \in A^c$ ($x \notin A$), this implies $S_{r_1}(x) \cap A = \emptyset \implies S_{r_1}(x) \subseteq A^c$.
    \item Moreover, since $A'$ is a closed set (by the previous theorem), its complement $(A')^c$ is an open set.
    \item Since $x \in (A')^c$, there exists $r_2 > 0$ such that $S_{r_2}(x) \subseteq (A')^c$.
    \item Define $r^* = \min(r_1, r_2) > 0$. Then:
    \begin{equation}
        S_{r^*}(x) \subseteq S_{r_1}(x) \subseteq A^c \quad \text{and} \quad S_{r^*}(x) \subseteq S_{r_2}(x) \subseteq (A')^c
    \end{equation}
    \item Consequently:
    \begin{equation}
        S_{r^*}(x) \subseteq A^c \cap (A')^c = (\overline{A})^c
    \end{equation}
    \item Since $x \in (\overline{A})^c$ was arbitrary, every point of $(\overline{A})^c$ is an interior point.
    \item Hence, $(\overline{A})^c$ is open, which proves that $\overline{A}$ is a \textbf{closed set}. \qed
\end{enumerate}
\end{exbox}

\newpage

\begin{exbox}{Proof of Properties (ii), (iii), and (iv)}
\textbf{Proof of Property (ii) [$A \subseteq B \implies \overline{A} \subseteq \overline{B}$]:}
\begin{enumerate}
    \item Let $x \in A'$. For every $r > 0$, $(S_r(x) \setminus \{x\}) \cap A \neq \emptyset$.
    \item Since $A \subseteq B$, we have $(S_r(x) \setminus \{x\}) \cap A \subseteq (S_r(x) \setminus \{x\}) \cap B$.
    \item Thus $(S_r(x) \setminus \{x\}) \cap B \neq \emptyset$, which means $x \in B'$. Hence $A' \subseteq B'$.
    \item Therefore:
    \begin{equation}
        \overline{A} = A \cup A' \subseteq B \cup B' = \overline{B} \quad \qed
    \end{equation}
\end{enumerate}

\textbf{Proof of Property (iii) [$A$ is closed $\iff \overline{A} = A$]:}
\begin{enumerate}
    \item ($\implies$) Suppose $A$ is closed. Then $A$ contains all its limit points, so $A' \subseteq A$.
    \begin{equation}
        \overline{A} = A \cup A' = A
    \end{equation}
    \item ($\impliedby$) Conversely, suppose $\overline{A} = A$.
    \item Since $\overline{A}$ is always a closed set (by Property (i)), it immediately follows that $A$ is closed. \qed
\end{enumerate}

\textbf{Proof of Property (iv) [$\overline{A}$ is the smallest closed set containing $A$]:}
\begin{enumerate}
    \item By Property (i), $\overline{A}$ is a closed set, and clearly $A \subseteq A \cup A' = \overline{A}$.
    \item Let $F$ be any closed set containing $A$ ($A \subseteq F$).
    \item By the monotonicity property (ii), $A \subseteq F \implies \overline{A} \subseteq \overline{F}$.
    \item Since $F$ is closed, by Property (iii), $\overline{F} = F$.
    \item Therefore, $\overline{A} \subseteq F$.
    \item Since this holds for every closed superset $F$, $\overline{A}$ is indeed the \textbf{smallest closed set containing $A$}. \qed
\end{enumerate}
\end{exbox}

\begin{exbox}{Proof of Properties (v), (vi), and (vii)}
\textbf{Proof of Property (v) [$\overline{A \cup B} = \overline{A} \cup \overline{B}$]:}
\begin{enumerate}
    \item Since $A \subseteq A \cup B$ and $B \subseteq A \cup B$, by monotonicity (ii):
    \begin{equation}
        \overline{A} \subseteq \overline{A \cup B} \quad \text{and} \quad \overline{B} \subseteq \overline{A \cup B} \implies \overline{A} \cup \overline{B} \subseteq \overline{A \cup B} \tag{1}
    \end{equation}
    \item On the other hand, $A \subseteq \overline{A}$ and $B \subseteq \overline{B} \implies A \cup B \subseteq \overline{A} \cup \overline{B}$.
    \item Since $\overline{A}$ and $\overline{B}$ are closed sets, their union $\overline{A} \cup \overline{B}$ is a closed set containing $A \cup B$.
    \item Since $\overline{A \cup B}$ is the \textbf{smallest} closed set containing $A \cup B$ (Property (iv)), we must have:
    \begin{equation}
        \overline{A \cup B} \subseteq \overline{A} \cup \overline{B} \tag{2}
    \end{equation}
    \item Combining (1) and (2) gives $\overline{A \cup B} = \overline{A} \cup \overline{B}$. \qed
\end{enumerate}

\textbf{Proof of Property (vi) [$\overline{A \cap B} \subseteq \overline{A} \cap \overline{B}$]:}
\begin{enumerate}
    \item Since $A \cap B \subseteq A \implies \overline{A \cap B} \subseteq \overline{A}$, and $A \cap B \subseteq B \implies \overline{A \cap B} \subseteq \overline{B}$.
    \item Therefore, $\overline{A \cap B} \subseteq \overline{A} \cap \overline{B}$. \qed
    \item \textbf{Counterexample for Strict Inclusion:} In $(\mathbb{R}, |\cdot|)$, let $A = (0, 1)$ and $B = (1, 2)$.
    $$A \cap B = \emptyset \implies \overline{A \cap B} = \overline{\emptyset} = \emptyset$$
    However, $\overline{A} = [0, 1]$ and $\overline{B} = [1, 2]$, so $\overline{A} \cap \overline{B} = [0, 1] \cap [1, 2] = \{1\} \neq \emptyset$.
\end{enumerate}

\textbf{Proof of Property (vii) [$\overline{\overline{A}} = \overline{A}$]:}
\begin{enumerate}
    \item Since $\overline{A}$ is a closed set (Property (i)), applying Property (iii) directly to the closed set $\overline{A}$ yields $\overline{\overline{A}} = \overline{A}$. \qed
\end{enumerate}
\end{exbox}

\subsection{Perfect Sets}

\begin{defbox}{Definition: Perfect Set}
Let $(X, d)$ be a metric space. A subset $A \subseteq X$ is called a \textbf{perfect set} if and only if:
\begin{enumerate}
    \item $A$ is a \textbf{closed set} ($A' \subseteq A$), and
    \item Every point of $A$ is a \textbf{limit point} of $A$ ($A \subseteq A'$).
\end{enumerate}
Equivalently, a subset $A$ is perfect if and only if it equals its own derived set:
\begin{equation}
    \mathbf{A = A'}
\end{equation}
In other words, a perfect set is a closed set with \textbf{no isolated points}.
\end{defbox}

\begin{exbox}{Examples of Perfect and Non-Perfect Sets}
\begin{enumerate}
    \item \textbf{Closed Intervals in $\mathbb{R}$:}
    Any closed interval $A = [a, b]$ with $a < b$ in $(\mathbb{R}, |\cdot|)$ is perfect because $A' = [a, b] = A$.
    
    \item \textbf{The Real Line $\mathbb{R}$ and Euclidean Space $\mathbb{R}^n$:}
    $\mathbb{R}$ is closed and every real number is a limit point of $\mathbb{R}$, so $\mathbb{R}' = \mathbb{R}$, making it perfect.
    
    \item \textbf{The Cantor Ternary Set $\mathcal{C}$:}
    The standard Cantor middle-thirds set is closed, totally disconnected, contains no isolated points, and $\mathcal{C}' = \mathcal{C}$. Thus, the Cantor set is a \textbf{perfect set} with Lebesgue measure $0$ and cardinality of the continuum ($\mathfrak{c}$).
    
    \item \textbf{Non-Perfect Set Example (Singletons and Discrete Sets):}
    The singleton $\{x\}$ in $\mathbb{R}$ is closed, but its derived set is empty $(\{x\}' = \emptyset \neq \{x\})$. Hence singletons are \textbf{not} perfect sets.
\end{enumerate}
\end{exbox}

\subsection{The Interior Operator and Its Fundamental Properties}

\begin{defbox}{Definition: Interior of a Set ($A^\circ$ / $\operatorname{Int}(A)$)}
Let $(X, d)$ be a metric space and $A \subseteq X$. The \textbf{interior} of $A$, denoted by $A^\circ$ or $\operatorname{Int}(A)$, is the set of all interior points of $A$:
\begin{equation}
    A^\circ = \operatorname{Int}(A) = \{ x \in A \mid \exists r > 0 \text{ such that } S_r(x) \subseteq A \}
\end{equation}
\end{defbox}

\begin{formulabox}{Theorem: Fundamental Properties of the Interior Operator}
Let $(X, d)$ be a metric space and $A, B \subseteq X$.
\begin{enumerate}
    \item[\textbf{(i)}] $A^\circ \subseteq A$.
    \item[\textbf{(ii)}] $A^\circ$ is an \textbf{open set} in $(X, d)$.
    \item[\textbf{(iii)}] $A$ is an open set if and only if $A^\circ = A$.
    \item[\textbf{(iv)}] If $B \subseteq A$ and $B$ is open, then $B \subseteq A^\circ$.
    \item[\textbf{(v)}] $A^\circ$ is the \textbf{largest open set} contained in $A$, i.e.,
    $$A^\circ = \bigcup \{ G \subseteq X \mid G \subseteq A \text{ and } G \text{ is open} \}$$
    \item[\textbf{(vi)}] $(A \cap B)^\circ = A^\circ \cap B^\circ$ and $A^\circ \cup B^\circ \subseteq (A \cup B)^\circ$.
    \item[\textbf{(vii)}] $(A^\circ)^\circ = A^\circ$ (Idempotence).
\end{enumerate}
\end{formulabox}

\subsubsection{Complete Rigorous Proofs for Interior Properties}

\begin{exbox}{Proofs of Interior Properties (i) through (v)}
\textbf{Proof of Property (i) [$A^\circ \subseteq A$]:}
\begin{itemize}
    \item By definition, $A^\circ = \{x \in A \mid \exists r > 0, S_r(x) \subseteq A\}$.
    \item Thus, if $x \in A^\circ$, then by definition $x \in A$. Therefore, $A^\circ \subseteq A$. \qed
\end{itemize}

\textbf{Proof of Property (ii) [$A^\circ$ is an Open Set]:}
\begin{enumerate}
    \item Let $x \in A^\circ$. By definition of an interior point, there exists $r > 0$ such that $S_r(x) \subseteq A$.
    \item We must show that there exists an open ball around $x$ entirely contained in $A^\circ$.
    \item In fact, we claim that $S_r(x) \subseteq A^\circ$.
    \item Let $y \in S_r(x)$ be any arbitrary point.
    \item Since every open sphere in a metric space is an open set (Theorem 2), there exists $\rho = r - d(x, y) > 0$ such that:
    \begin{equation}
        S_\rho(y) \subseteq S_r(x)
    \end{equation}
    \item Since $S_r(x) \subseteq A$, it follows that $S_\rho(y) \subseteq A$.
    \item This means that $y$ is an interior point of $A$, so $y \in A^\circ$.
    \item Since $y \in S_r(x)$ was arbitrary, $S_r(x) \subseteq A^\circ$.
    \item Hence, $x$ is an interior point of $A^\circ$. Since $x \in A^\circ$ was arbitrary, $A^\circ$ is an \textbf{open set}. \qed
\end{enumerate}

\textbf{Proof of Property (iii) [$A$ is open $\iff A^\circ = A$]:}
\begin{enumerate}
    \item ($\implies$) Suppose $A$ is open. By Property (i), $A^\circ \subseteq A$.
    \item Now let $x \in A$. Since $A$ is open, by definition there exists $r > 0$ such that $S_r(x) \subseteq A \implies x \in A^\circ$.
    \item Thus $A \subseteq A^\circ$. Combining both inclusions gives $A^\circ = A$.
    \item ($\impliedby$) Conversely, suppose $A^\circ = A$.
    \item Since $A^\circ$ is open by Property (ii), it immediately follows that $A$ is an open set. \qed
\end{enumerate}

\textbf{Proof of Properties (iv) and (v) [$A^\circ$ is the largest open set contained in $A$]:}
\begin{enumerate}
    \item Let $B \subseteq A$ with $B$ being open.
    \item Let $x \in B$. Since $B$ is open, $\exists r > 0$ such that $S_r(x) \subseteq B \subseteq A$.
    \item This implies $S_r(x) \subseteq A$, so $x \in A^\circ$. Hence $B \subseteq A^\circ$.
    \item Since $A^\circ$ is itself an open set contained in $A$ (Properties (i) and (ii)), and contains any other open set $B \subseteq A$, $A^\circ$ is the \textbf{largest open set contained in $A$}. \qed
\end{enumerate}
\end{exbox}

\subsection{Duality Between Interior, Closure, and Boundary}

\begin{formulabox}{Complement Duality Relations}
For any subset $A$ of a metric space $(X, d)$:
\begin{equation}
    (A^c)^\circ = (\overline{A})^c \qquad \text{and} \qquad \overline{A^c} = (A^\circ)^c
\end{equation}
\textbf{Boundary (Frontier) of a Set $\partial A$:}
\begin{equation}
    \partial A = \operatorname{Bd}(A) = \overline{A} \setminus A^\circ = \overline{A} \cap \overline{A^c}
\end{equation}
\end{formulabox}

\subsection{Metric Subspaces and Relative Topologies}

\begin{defbox}{Definition: Metric Subspace $(Y, d_Y)$}
Let $(X, d)$ be a metric space, and let $Y$ be a non-empty subset of $X$ ($Y \subseteq X, Y \neq \emptyset$).
\begin{itemize}
    \item Define the distance function $d_Y: Y \times Y \to \mathbb{R}$ by restricting $d$ to $Y$:
    \begin{equation}
        d_Y(x, y) = d(x, y), \quad \forall x, y \in Y
    \end{equation}
    \item Since $d$ satisfies axioms $[M1]-[M4]$ on all of $X$, the restriction $d_Y$ automatically satisfies all metric axioms on $Y$.
    \item The pair $(Y, d_Y)$ (or simply $(Y, d)$) is called a \textbf{Metric Subspace} of $(X, d)$.
\end{itemize}
\end{defbox}

\begin{formulabox}{Open Spheres in a Metric Subspace}
For any $y_0 \in Y$ and $r > 0$, the open sphere in the subspace $(Y, d_Y)$ is:
\begin{equation}
    S_r^Y(y_0) = \{ y \in Y \mid d_Y(y, y_0) < r \} = \{ y \in X \mid d(y, y_0) < r \} \cap Y = S_r^X(y_0) \cap Y
\end{equation}
That is, \textbf{an open ball in $Y$ is simply the intersection of an open ball in $X$ with $Y$}.
\end{formulabox}

\begin{exbox}{Example: Relative Openness vs. Global Openness}
Consider the Euclidean plane $\mathbb{R}^2$ with standard metric $d_2$. Let $Y$ be the $x$-axis:
$$Y = \{ (x, 0) \in \mathbb{R}^2 \mid x \in \mathbb{R} \}$$
Let $E = \{ (x, 0) \in \mathbb{R}^2 \mid -1 < x < 1 \} = (-1, 1) \times \{0\}$.
\begin{itemize}
    \item In the subspace $(Y, d_Y)$, $E = S_1^Y((0,0))$ is an open ball of radius 1 centered at $(0,0)$. Thus, \textbf{$E$ is open relative to $Y$}.
    \item In the full space $(\mathbb{R}^2, d_2)$, every 2D open disk $S_r^{\mathbb{R}^2}((0,0))$ contains points with non-zero $y$-coordinates (points off the $x$-axis). Thus $S_r^{\mathbb{R}^2}((0,0)) \not\subseteq E$ for any $r > 0$.
    \item Therefore, \textbf{$E$ is NOT open in $\mathbb{R}^2$}.
\end{itemize}
\end{exbox}

\subsection{Fundamental Theorems on Subspace Topology}

\begin{exbox}{Theorem: Characterization of Relative Open Sets}
\textbf{Theorem:} Let $(Y, d)$ be a metric subspace of $(X, d)$. A subset $E \subseteq Y$ is \textbf{open relative to $Y$} if and only if there exists an open set $G \subseteq X$ such that:
\begin{equation}
    E = Y \cap G
\end{equation}

\textbf{Proof:}
\begin{enumerate}
    \item \textbf{($\implies$) Direct Part:} Assume $E$ is open relative to $Y$.
    \begin{itemize}
        \item For each point $p \in E$, since $E$ is open in $(Y, d)$, there exists a radius $r_p > 0$ such that the subspace open ball centered at $p$ satisfies:
        \begin{equation}
            S_{r_p}^Y(p) \subseteq E
        \end{equation}
        \item Recall that $S_{r_p}^Y(p) = \{ q \in Y \mid d(p, q) < r_p \} = S_{r_p}^X(p) \cap Y$, where $S_{r_p}^X(p) = \{ q \in X \mid d(p, q) < r_p \}$.
        \item Let $V_p = S_{r_p}^X(p)$ be the open ball in $X$. By Theorem 2, each $V_p$ is an open set in $(X, d)$.
        \item Define $G = \bigcup_{p \in E} V_p$.
        \item Since each $V_p$ is open in $X$ and the arbitrary union of open sets is open (Theorem 3), $G$ is an \textbf{open set in $X$}.
        \item We now verify that $E = Y \cap G$:
        \begin{itemize}
            \item Since $p \in V_p$ for every $p \in E$, we have $E \subseteq \bigcup_{p \in E} V_p = G$. Since $E \subseteq Y$, it follows that $E \subseteq Y \cap G$.
            \item Conversely, let $y \in Y \cap G$. Then $y \in Y$ and $y \in \bigcup_{p \in E} V_p$.
            \item Thus $y \in V_{p_0} = S_{r_{p_0}}^X(p_0)$ for some specific $p_0 \in E$.
            \item Since $y \in Y$, we have $y \in S_{r_{p_0}}^X(p_0) \cap Y = S_{r_{p_0}}^Y(p_0) \subseteq E$.
            \item Hence $y \in E$, which proves $Y \cap G \subseteq E$.
        \end{itemize}
        \item Combining both inclusions yields $E = Y \cap G$.
    \end{itemize}
    
    \item \textbf{($\impliedby$) Converse Part:} Assume $E = Y \cap G$ for some open set $G$ in $(X, d)$.
    \begin{itemize}
        \item Let $p \in E = Y \cap G$. Then $p \in Y$ and $p \in G$.
        \item Since $G$ is open in $X$, there exists an open ball $S_r^X(p)$ in $X$ with radius $r > 0$ such that $S_r^X(p) \subseteq G$.
        \item Then the subspace open ball in $Y$ centered at $p$ satisfies:
        \begin{equation}
            S_r^Y(p) = S_r^X(p) \cap Y \subseteq G \cap Y = E
        \end{equation}
        \item Thus $S_r^Y(p) \subseteq E$, which proves that $p$ is an interior point of $E$ in the metric space $(Y, d_Y)$.
        \item Since $p \in E$ was arbitrary, $E$ is \textbf{open relative to $Y$}. \qed
    \end{itemize}
\end{enumerate}
\end{exbox}

\begin{exbox}{Theorem: Characterization of Relative Closed Sets}
\textbf{Theorem:} Let $(Y, d)$ be a metric subspace of $(X, d)$. A subset $K \subseteq Y$ is \textbf{closed relative to $Y$} if and only if there exists a closed set $F \subseteq X$ such that:
\begin{equation}
    K = Y \cap F
\end{equation}

\textbf{Proof:}
\begin{enumerate}
    \item By definition, $K$ is closed in $Y$ if and only if its relative complement $Y \setminus K$ is an \textbf{open set in $Y$}.
    \item By the Relative Open Set Theorem, $Y \setminus K$ is open in $Y$ if and only if there exists an open set $G \subseteq X$ such that:
    \begin{equation}
        Y \setminus K = Y \cap G
    \end{equation}
    \item Taking complements in $Y$:
    \begin{align}
        K &= Y \setminus (Y \setminus K) = Y \setminus (Y \cap G) \notag \\
        &= Y \cap (X \setminus G)
    \end{align}
    \item Let $F = X \setminus G = G^c$.
    \item Since $G$ is an open set in $X$, its complement $F = X \setminus G$ is a \textbf{closed set in $X$}.
    \item Therefore, $K = Y \cap F$ for the closed set $F \subseteq X$.
    \item Conversely, if $K = Y \cap F$ for some closed set $F \subseteq X$, then:
    \begin{equation}
        Y \setminus K = Y \setminus (Y \cap F) = Y \cap (X \setminus F) = Y \cap G
    \end{equation}
    where $G = X \setminus F$ is open in $X$.
    \item Thus $Y \setminus K$ is open in $Y$, which means $K$ is \textbf{closed relative to $Y$}. \qed
\end{enumerate}
\end{exbox}

\begin{formulabox}{Important Corollaries on Subspace Open/Closed Sets}
\begin{enumerate}
    \item \textbf{When $Y$ is Open in $X$:}
    If the subspace $Y$ is an open subset of $X$, then a subset $E \subseteq Y$ is open in $Y$ if and only if $E$ is \textbf{open in $X$}.
    \item \textbf{When $Y$ is Closed in $X$:}
    If the subspace $Y$ is a closed subset of $X$, then a subset $K \subseteq Y$ is closed in $Y$ if and only if $K$ is \textbf{closed in $X$}.
\end{enumerate}
\end{formulabox}

\section{Module 6: Master Quick Revision Cheat Sheet}

\begin{formulabox}{1. Metric Axioms, Norms, and Standard Metrics}
\begin{itemize}
    \item \textbf{Metric Space Axioms $(X, d)$:} $\forall x, y, z \in X$:
    \begin{enumerate}
        \item \textbf{[M1] Non-negativity:} $d(x, y) \ge 0$.
        \item \textbf{[M2] Identity of Indiscernibles:} $d(x, y) = 0 \iff x = y$.
        \item \textbf{[M3] Symmetry:} $d(x, y) = d(y, x)$.
        \item \textbf{[M4] Triangle Inequality:} $d(x, z) \le d(x, y) + d(y, z)$.
    \end{enumerate}
    \item \textbf{Standard Metrics on $\mathbb{R}^n$:}
    \begin{itemize}
        \item Euclidean Metric ($L_2$): $d_2(x, y) = \sqrt{\sum_{i=1}^n (x_i - y_i)^2}$
        \item Taxicab / Manhattan Metric ($L_1$): $d_1(x, y) = \sum_{i=1}^n |x_i - y_i|$
        \item Maximum / Chebyshev Metric ($L_\infty$): $d_\infty(x, y) = \max_{1 \le i \le n} |x_i - y_i|$
        \item Discrete Metric: $d(x, y) = 0$ if $x = y$, and $d(x, y) = 1$ if $x \neq y$.
        \item Semi-metric (Pseudo-metric): $[M1], [M3], [M4]$ hold, but $d(x, y) = 0 \not\implies x = y$ (e.g., $d(x, y) = |x^2 - y^2|$).
    \end{itemize}
    \item \textbf{Matrix Frobenius Metric on $M_{m \times n}(\mathbb{R})$:}
    $d_F(A, B) = \|A - B\|_F = \sqrt{\sum_{i,j} (a_{ij} - b_{ij})^2} = \sqrt{\operatorname{Tr}((A - B)^T (A - B))}$.
\end{itemize}
\end{formulabox}

\begin{defbox}{2. Open Sets, Spheres, and Neighborhoods}
\begin{itemize}
    \item \textbf{Open Sphere (Ball):} $S_r(x_0) = B(x_0, r) = \{x \in X \mid d(x, x_0) < r\}$.
    \item \textbf{Closed Sphere (Ball):} $\overline{S}_r(x_0) = \bar{B}(x_0, r) = \{x \in X \mid d(x, x_0) \le r\}$.
    \item \textbf{Interior Point:} $x \in A$ is an interior point of $A$ if $\exists r > 0$ such that $S_r(x) \subseteq A$.
    \item \textbf{Open Set:} $A \subseteq X$ is open if every point of $A$ is an interior point of $A$ (i.e., $A = A^\circ$).
    \item \textbf{Fundamental Theorem on Open Balls:} Every open sphere $S_r(x_0)$ in $(X, d)$ is an \textbf{open set} (choose $\rho = r - d(x, x_0) > 0$).
    \item \textbf{Topological Properties of Open Sets:}
    \begin{itemize}
        \item $\emptyset$ and $X$ are open sets.
        \item The \textbf{arbitrary union} of open sets is open: $\bigcup_{\lambda \in \Lambda} U_\lambda$ is open.
        \item The \textbf{finite intersection} of open sets is open: $\bigcap_{i=1}^n U_i$ is open.
        \item Infinite intersections of open sets need not be open (Counterexample: $\bigcap_{n=1}^\infty \left(-\frac{1}{n}, \frac{1}{n}\right) = \{0\}$).
    \end{itemize}
\end{itemize}
\end{defbox}

\begin{exbox}{3. Limit Points, Closures, Perfect Sets, and Closed Sets}
\begin{itemize}
    \item \textbf{Limit Point (Accumulation Point):} $p \in X$ is a limit point of $A$ if $\forall r > 0, \; (S_r(p) \setminus \{p\}) \cap A \neq \emptyset$.
    \item \textbf{Derived Set $A'$:} Set of all limit points of $A$. \textbf{Theorem:} Every derived set $A'$ is a \textbf{closed set} ($(A')' \subseteq A'$).
    \item \textbf{Closed Set Characterization:} $F$ is closed $\iff F^c$ is open $\iff F' \subseteq F \iff \overline{F} = F$.
    \item \textbf{Closure of a Set $\overline{A} = A \cup A'$:}
    \begin{itemize}
        \item $\overline{A}$ is closed.
        \item $A \subseteq B \implies \overline{A} \subseteq \overline{B}$.
        \item $\overline{A}$ is the smallest closed set containing $A$ ($\overline{A} = \bigcap \{F \supseteq A \mid F \text{ closed}\}$).
        \item $\overline{A \cup B} = \overline{A} \cup \overline{B}$, and $\overline{A \cap B} \subseteq \overline{A} \cap \overline{B}$.
        \item Idempotence: $\overline{\overline{A}} = \overline{A}$.
    \end{itemize}
    \item \textbf{Perfect Set:} $A$ is closed and has no isolated points $\iff A = A'$. (e.g., $[a, b]$, $\mathbb{R}^n$, Cantor Set $\mathcal{C}$).
\end{itemize}
\end{exbox}

\begin{formulabox}{4. Interior Operator, Boundary, and Subspace Topologies}
\begin{itemize}
    \item \textbf{Interior of a Set $A^\circ = \operatorname{Int}(A)$:}
    \begin{itemize}
        \item $A^\circ \subseteq A$, and $A^\circ$ is an open set.
        \item $A$ is open $\iff A^\circ = A$.
        \item $A^\circ$ is the largest open set contained in $A$ ($A^\circ = \bigcup \{G \subseteq A \mid G \text{ open}\}$).
        \item $(A \cap B)^\circ = A^\circ \cap B^\circ$, and $(A^\circ)^\circ = A^\circ$.
    \end{itemize}
    \item \textbf{Duality \& Boundary:}
    \begin{equation*}
        (A^c)^\circ = (\overline{A})^c, \qquad \overline{A^c} = (A^\circ)^c, \qquad \partial A = \overline{A} \setminus A^\circ = \overline{A} \cap \overline{A^c}
    \end{equation*}
    \item \textbf{Metric Subspaces $(Y, d_Y)$ where $d_Y = d|_{Y \times Y}$:}
    \begin{itemize}
        \item Open Balls in Subspace: $S_r^Y(y) = S_r^X(y) \cap Y$.
        \item \textbf{Relative Open Set Theorem:} $E \subseteq Y$ is open in $Y \iff \exists \text{ open } G \subseteq X \text{ such that } E = Y \cap G$.
        \item \textbf{Relative Closed Set Theorem:} $K \subseteq Y$ is closed in $Y \iff \exists \text{ closed } F \subseteq X \text{ such that } K = Y \cap F$.
    \end{itemize}
\end{itemize}
\end{formulabox}
