\chapter{Constrained Motion on Vertical Curves}

\section{Motion on a Smooth Vertical Circle}

\begin{theorembox}{Vertical Circle Governing Equations}
For horizontal projection velocity $u$ from lowest point $A(\theta = 0)$ of smooth vertical circle of radius $a$:
1. Velocity at angle $\theta$: $v^2 = u^2 - 2ga(1 - \cos\theta)$
2. Normal Reaction: $N = \frac{m}{a}[u^2 - 2ga + 3ga\cos\theta]$
3. Complete Loop Condition: $u \ge \sqrt{5ga}$
4. Oscillation Condition: $u \le \sqrt{2ga}$

\end{theorembox}

\begin{proofbox}{Proof of SHM About Lowest Point for Small Oscillations}
For small oscillations about lowest point $A(\theta = 0)$, $\sin\theta \approx \theta$.
Equation of motion along arc $s = a\theta$:
\[
\ddot{s} = -g \sin\theta \approx -g \left(\frac{s}{a}\right) = -\left(\frac{g}{a}\right) s
\]
This is exact **Simple Harmonic Motion** $\ddot{s} = -\mu s$ with $\mu = g/a$. Periodic time:
\begin{equation}
T = \frac{2\pi}{\sqrt{\mu}} = 2\pi \sqrt{\frac{a}{g}}
\end{equation}
\end{proofbox}

\section{Motion on a Smooth Vertical Cycloid}

\begin{theorembox}{Cycloidal Pendulum}
For motion along smooth cycloid $s = 4a\sin\psi$ (vertex at lowest point):
Equation of motion: $\ddot{s} = -g\sin\psi = -\frac{g}{4a} s$.
Exact \textbf{Simple Harmonic Motion} with periodic time:
\begin{equation}
T = 4\pi \sqrt{\frac{a}{g}}
\end{equation}
The motion is \textbf{isochronous} (period is independent of amplitude).

\begin{proofbox}{Time Taken to Reach Different Heights on Cycloid}
Let particle start from rest at height corresponding to arc length $s = s_0$ ($\psi = \psi_0$).
Integrating equation of motion $\ddot{s} = -\omega^2 s$ (where $\omega = \frac{1}{2}\sqrt{g/a}$):
\[
v^2 = \left(\frac{ds}{dt}\right)^2 = \omega^2 (s_0^2 - s^2) \implies \frac{ds}{\sqrt{s_0^2 - s^2}} = -\omega \, dt
\]
Time $t$ taken to fall from $s_0$ to arc length $s$:
\[
t = \frac{1}{\omega} \cos^{-1}\left(\frac{s}{s_0}\right) = 2\sqrt{\frac{a}{g}} \cos^{-1}\left(\frac{\sin\psi}{\sin\psi_0}\right)
\]
Time taken to reach the vertex ($s = 0$): $t_{\text{vertex}} = \frac{\pi}{2\omega} = \pi \sqrt{\frac{a}{g}}$.
\end{proofbox}
